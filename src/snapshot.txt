const AuthContext = createContext<AuthContextType | undefined>(undefined);

const SESSION_KEY = "octomation_session";
const API_URL = "http://127.0.0.1:4002/iam";

function encryptSession(data: unknown): string {
  const serialized = JSON.stringify(data);

  if (window.api?.auth?.encrypt) {
    try {
      return window.api.auth.encrypt(serialized);
    } catch {
      // Usa el fallback de serialización.
    }
  }

  return btoa(unescape(encodeURIComponent(serialized)));
}

function decryptSession(encrypted: string): any {
  if (window.api?.auth?.decrypt) {
    try {
      return JSON.parse(window.api.auth.decrypt(encrypted));
    } catch {
      // Intenta recuperar la sesión con el formato de fallback.
    }
  }

  try {
    return JSON.parse(
      decodeURIComponent(escape(atob(encrypted)))
    );
  } catch {
    return null;
  }
}

function decodeJwtPayload(token: string): Record<string, unknown> {
  const parts = token.split(".");

  if (parts.length !== 3) {
    throw new Error("El servidor devolvió un JWT inválido");
  }

  const base64 = parts[1]
    .replace(/-/g, "+")
    .replace(/_/g, "/");

  const padded = base64.padEnd(
    base64.length + ((4 - base64.length % 4) % 4),
    "="
  );

  const bytes = Uint8Array.from(
    atob(padded),
    (char) => char.charCodeAt(0)
  );

  const payload = new TextDecoder().decode(bytes);

  return JSON.parse(payload);
}

export function AuthProvider({
  children
}: {
  children: ReactNode;
}) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    try {
      const stored = localStorage.getItem(SESSION_KEY);

      if (stored) {
        const session = decryptSession(stored);
        const token = session?.user?.token;

        if (typeof token === "string") {
          const payload = decodeJwtPayload(token);

          if (
            typeof payload.exp === "number" &&
            payload.exp * 1000 <= Date.now()
          ) {
            localStorage.removeItem(SESSION_KEY);
          } else {
            setUser(session.user);
          }
        } else {
          localStorage.removeItem(SESSION_KEY);
        }
      }
    } catch {
      localStorage.removeItem(SESSION_KEY);
    } finally {
      setLoading(false);
    }
  }, []);

  const login = async (email: string, password: string) => {
    const response = await fetch(`${API_URL}/login`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json"
      },
      body: JSON.stringify({
        email,
        password
      })
    });

    const data = await response.json().catch(() => null);

    if (!response.ok) {
      throw new Error(
        data?.message || "No se pudo iniciar sesión"
      );
    }

    const token = data?.token;

    if (typeof token !== "string") {
      throw new Error(
        "La respuesta del servidor no contiene token"
      );
    }

    const payload = decodeJwtPayload(token);

    if (
      typeof payload.sub !== "string" &&
      typeof payload.sub !== "number"
    ) {
      throw new Error("El JWT no contiene un identificador válido");
    }

    if (
      typeof payload.email !== "string" ||
      typeof payload.fullName !== "string" ||
      !Array.isArray(payload.roles) ||
      !Array.isArray(payload.permissions)
    ) {
      throw new Error("El payload del JWT tiene un formato inválido");
    }

    const sessionUser: User = {
      id: String(payload.sub),
      email: payload.email,
      fullName: payload.fullName,
      roles: payload.roles,
      permissions: payload.permissions,
      token
    };

    console.log(sessionUser) 

    localStorage.setItem(
      SESSION_KEY,
      encryptSession({ user: sessionUser })
    );

    setUser(sessionUser);
  };

  const logout = () => {
    setUser(null);
    localStorage.removeItem(SESSION_KEY);
  };

  return (
    <AuthContext.Provider
      value={{ user, loading, login, logout }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);

  if (!context) {
    throw new Error(
      "useAuth debe usarse dentro de AuthProvider"
    );
  }

  return context;
}

  useEffect(() => {
    if (!window.api) return;

    window.api.onPortUpdate((port) => {
      if (port) {
        window.API_PORT = port;
        setApiBase(`http://127.0.0.1:${port}`);
        setResult(`Servidor FastAPI listo en http://127.0.0.1:${port}`);
      }
      setStatus("online");
    });

    window.api.onStatusUpdate((s) => {
      setStatus(s.running ? "online" : "offline");
      if (!s.running) setPid("-");
    });

    window.api.onPidUpdate((p) => {
      setPid(p ? String(p) : "-");
    });

    window.api.getFastApiPort().then((port) => {
      if (port) {
        window.API_PORT = port;
        setApiBase(`http://127.0.0.1:${port}`);
        setStatus("online");
      }
    });
  }, []);

  const getBase = async () => {
    if (apiBase) return apiBase;
    const port = await window.api.getFastApiPort();
    const base = `http://127.0.0.1:${port || 8000}`;
    setApiBase(base);
    return base;
  };

  const callHealth = async () => {
    try {
      const base = await getBase();
      const res = await fetch(`${base}/health`);
      if (!res.ok) throw new Error(`Error HTTP: ${res.status}`);

      setResult(JSON.stringify(data, null, 2));
